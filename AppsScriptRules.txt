const DRIVE_ROOT_FOLDER_ID = '1i776DvoTe_7ZWCaK7PCslQeCVLo66aZ4';
const DRIVE_HTML_FOLDER_ID = '1yoE9-0PVarS3zsONpFW26cZg-0ieBNhj';
const DRIVE_WAV_FOLDER_ID  = '1p18_RKzQCNBObMkDnUR1_E3C7KOl_eWa';

function doPost(e) {
  try {
    var payload = JSON.parse((e && e.postData && e.postData.contents) ? e.postData.contents : '{}');
    var rec = payload.record || {};
    var archiveId = String(rec.archiveId || ('ARC-' + new Date().getTime())).replace(/[^a-zA-Z0-9_-]/g, '');
    var handle = (rec.creatorMetadata && rec.creatorMetadata.handle) || rec.authorHandle || '';

    // Exige formato de ID Único de 8 dígitos (@12345678)
    if (!/^@\d{8}$/.test(String(handle).trim())) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'INVALID_8_DIGIT_ID' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Salva o arquivo .HTML interativo na subpasta HTML do Google Drive
    if (payload.compiledHtmlContent) {
      var htmlFolder = DriveApp.getFolderById(DRIVE_HTML_FOLDER_ID);
      htmlFolder.createFile(archiveId + '.html', payload.compiledHtmlContent, MimeType.HTML);
    }

    // Salva o áudio .WAV em base64 (se enviado) na subpasta WAV do Google Drive
    if (rec.generatedWavBase64 && String(rec.generatedWavBase64).indexOf('base64,') !== -1) {
      var b64 = String(rec.generatedWavBase64).split('base64,')[1];
      var wavBlob = Utilities.newBlob(Utilities.base64Decode(b64), 'audio/wav', archiveId + '.wav');
      var wavFolder = DriveApp.getFolderById(DRIVE_WAV_FOLDER_ID);
      wavFolder.createFile(wavBlob);
    }

    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      stage: 'DOORKEEPER_SEALED_FIREBASE_RTDB',
      archiveId: archiveId
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
